# Dashboard API Service

The Dashboard API provides type-safe access to code analysis reports generated by Python analyzers.

## Features

- **Type-safe**: Full TypeScript type definitions matching Python dataclasses
- **Graceful degradation**: Missing reports return `null` instead of throwing errors
- **Validation**: Validates report structure before returning data
- **Error handling**: Detailed error messages logged to console
- **Parallel loading**: Loads all reports concurrently for better performance

## Usage

### Load All Reports

```typescript
import { dashboardApi } from '@/features/dashboard';

const result = await dashboardApi.loadAllReports('/path/to/outputs');

// Access individual reports (may be null if missing)
const { quality, coverage, dependencies } = result.data;

// Check for errors
if (result.errors.length > 0) {
  console.error('Some reports failed to load:', result.errors);
}

// Use the data
if (quality) {
  console.log(`Found ${quality.total_issues} quality issues`);
}

if (coverage) {
  console.log(`Test coverage: ${coverage.coverage_percentage.toFixed(1)}%`);
}

if (dependencies) {
  console.log(`Circular dependencies: ${dependencies.circular_dependencies.length}`);
}
```

### Load Individual Reports

```typescript
import { dashboardApi } from '@/features/dashboard';

// Load quality report
const quality = await dashboardApi.loadQualityReport('/path/to/outputs');
if (quality) {
  console.log(`Total issues: ${quality.total_issues}`);
  console.log('By severity:', quality.issues_by_severity);
  console.log('By category:', quality.issues_by_category);
}

// Load coverage report
const coverage = await dashboardApi.loadCoverageReport('/path/to/outputs');
if (coverage) {
  console.log(`Coverage: ${coverage.coverage_percentage}%`);
  console.log(`Tested: ${coverage.tested_functions}/${coverage.total_functions}`);
}

// Load dependency report
const dependencies = await dashboardApi.loadDependencyReport('/path/to/outputs');
if (dependencies) {
  console.log(`Total deps: ${dependencies.total_dependencies}`);
  console.log('Circular deps:', dependencies.circular_dependencies);
}
```

### React Hook Example

```typescript
import { useState, useEffect } from 'react';
import { dashboardApi, type PythonAnalyzerData } from '@/features/dashboard';

function useDashboardData(outputsPath: string) {
  const [data, setData] = useState<PythonAnalyzerData | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    async function loadData() {
      try {
        setLoading(true);
        const result = await dashboardApi.loadAllReports(outputsPath);

        if (result.errors.length > 0) {
          console.warn('Some reports failed to load:', result.errors);
        }

        setData(result.data);
      } catch (err) {
        setError(err instanceof Error ? err.message : 'Failed to load reports');
      } finally {
        setLoading(false);
      }
    }

    loadData();
  }, [outputsPath]);

  return { data, loading, error };
}

// Usage in component
function DashboardPage() {
  const { data, loading, error } = useDashboardData('./outputs');

  if (loading) return <div>Loading...</div>;
  if (error) return <div>Error: {error}</div>;
  if (!data) return <div>No data available</div>;

  return (
    <div>
      {data.quality && <QualitySection data={data.quality} />}
      {data.coverage && <CoverageSection data={data.coverage} />}
      {data.dependencies && <DependencySection data={data.dependencies} />}
    </div>
  );
}
```

## File Structure

The API expects reports to be located at:

```
outputs/
├── quality/
│   └── quality_report.json
├── coverage/
│   └── coverage_report.json
└── dependencies/
    └── dependency_report.json
```

## Type Definitions

### Quality Report

```typescript
interface PythonQualityReport {
  total_files_scanned: number;
  total_issues: number;
  issues_by_severity: Record<string, number>;
  issues_by_category: Record<string, number>;
  issues: PythonQualityIssue[];
}

interface PythonQualityIssue {
  severity: 'error' | 'warning' | 'info';
  category: 'code_smell' | 'security' | 'documentation' | 'best_practice';
  rule_id: string;
  message: string;
  file_path: string;
  line_number: number;
  code_snippet: string | null;
  suggestion: string | null;
}
```

### Coverage Report

```typescript
interface PythonCoverageReport {
  total_functions: number;
  tested_functions: number;
  untested_functions: number;
  coverage_percentage: number;
  functions: PythonFunctionInfo[];
  untested_by_file: Record<string, PythonFunctionInfo[]>;
}

interface PythonFunctionInfo {
  name: string;
  file_path: string;
  line_number: number;
  is_async: boolean;
  is_tested: boolean;
  test_file: string | null;
}
```

### Dependency Report

```typescript
interface PythonDependencyReport {
  total_dependencies: number;
  external_dependencies: number;
  internal_dependencies: number;
  dependencies_by_file: Record<string, PythonDependencyInfo[]>;
  dependency_graph: Record<string, string[]>;
  circular_dependencies: string[][];
  unused_dependencies: string[];
}

interface PythonDependencyInfo {
  package: string;
  import_type: 'static' | 'dynamic' | 'require' | 'type_only';
  file_path: string;
  line_number: number;
  is_external: boolean;
}
```

## Error Handling

The API distinguishes between two types of errors:

1. **Missing files**: Return `null` (graceful degradation)
2. **Invalid data**: Throw error with detailed message

```typescript
try {
  const result = await dashboardApi.loadAllReports('./outputs');

  // Missing files don't cause errors
  if (!result.data.quality) {
    console.log('Quality report not available');
  }

  // Validation errors are collected
  if (result.errors.length > 0) {
    result.errors.forEach(err => {
      console.error(`${err.reportType}: ${err.error} at ${err.path}`);
    });
  }
} catch (error) {
  // This would be a network error or catastrophic failure
  console.error('Fatal error loading reports:', error);
}
```

## Environment Support

The API works in both Node.js and browser environments:

- **Node.js**: Uses `fs/promises` to read files from disk
- **Browser**: Uses `fetch` to load files via HTTP

The environment is detected automatically using `typeof window`.
